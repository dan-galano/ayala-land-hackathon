  <action>
    Create `client/components/kpi/KpiSection.tsx` as a "use client" component that fetches `/api/summary` once on mount and distributes the Summary data to all child components via props.

    ```tsx
    "use client"

    import { useEffect, useState } from "react"
    import type { Summary } from "@/lib/types"
    import { fetchSummary } from "@/lib/api"
    import { KpiCard } from "./KpiCard"
    import { SbtiIndicator } from "./SbtiIndicator"
    import { SptProgressBar } from "./SptProgressBar"

    export function KpiSection() {
      const [summary, setSummary] = useState<Summary | null>(null)
      const [error, setError] = useState<string | null>(null)

      useEffect(() => {
        fetchSummary()
          .then(setSummary)
          .catch((e) => setError(e.message))
      }, [])

      if (error) {
        return (
          <div className="rounded-lg border border-red-800 bg-red-900/20 p-4 text-red-400">
            Failed to load KPI data: {error}
          </div>
        )
      }

      if (!summary) {
        return (
          <div className="text-slate-500 text-sm">Loading KPI data...</div>
        )
      }

      return (
        <section className="space-y-4">
          <div className="grid grid-cols-4 gap-4">
            <KpiCard
              title="Total Consumption"
              value={`${summary.totalKwh.toLocaleString()} kWh`}
            />
            <KpiCard
              title="Scope 3 Tenant Emissions"
              value={`${summary.totalCo2e.toLocaleString()} kg CO₂e`}
            />
            <KpiCard
              title="Units Over Threshold"
              value={`${summary.unitsOverThreshold}`}
              subtitle={`of ${summary.unitCount} total units`}
            />
            <KpiCard
              title="Energy Savings vs Baseline"
              value={`${summary.savingsPercent.toFixed(1)}%`}
            />
          </div>
          <div className="grid grid-cols-2 gap-4">
            <SbtiIndicator
              progress={summary.sbtiProgress}
              savingsPercent={summary.savingsPercent}
            />
            <SptProgressBar
              progress={summary.sptProgress}
              savingsPercent={summary.savingsPercent}
            />
          </div>
        </section>
      )
    }
    ```

    CRITICAL requirements verified in this component:
    - VIZ-01: Four KPI cards with total kWh, CO2e, units over threshold, savings %
    - ESG-01: CO2e card title is exactly "Scope 3 Tenant Emissions"
    - ESG-02: SbtiIndicator receives sbtiProgress and savingsPercent
    - FIN-01: SptProgressBar receives sptProgress and savingsPercent
    - Coherence: ALL values from a single `summary` object — no recomputation
    - The component accepts no props (it owns the fetch). Phase 4 will refactor to accept `summary` as a prop when lifting state to the page level for Promise.all across all endpoints. Design consideration: KpiSection can be trivially refactored to accept optional `summary?: Summary` prop — if provided, skip the internal fetch. Do NOT implement this yet; Phase 4 handles it.

    Loading state is a simple text placeholder — per research, "demo starts from a loaded state; judges don't care about the 200ms loading flash." Error state is visible for debugging during development.
